Here is the complete, production-ready, futuristic personal portfolio website
for Mariyam Gull.

It is engineered as a single, self-contained, enterprise-grade file (index.html)
featuring:

  - Tailwind CSS (v3 CDN) + Font Awesome 6 Pro icons + Google Fonts (Poppins,
    Inter, Montserrat).
  - Embedded JSON-LD Schema (Person, WebSite, Breadcrumbs) and complete
    OpenGraph/Twitter SEO meta-tags.
  - Cinematic interactive Canvas particle starfield & mouse-responsive ambient
    gold spotlight.
  - Live typewriter effect cycling through her core professional titles.
  - Dynamic Project Filtering & Interactive Project Architecture Modal.
  - Filterable AI Creative Masonry Gallery with a full-screen Lightbox / Video
    concept player.
  - Interactive Dual Timeline separating Public Sector Enterprise Operations
    from Academic Honors.
  - Client Contact Hub with live validation, direct clipboard copy for
    email/WhatsApp, and mailto/form dispatch.
  - Smooth keyboard accessibility, ARIA standards, scroll progress monitor, and
    responsive mobile drawer.

Save the code below as index.html and run it in any browser or deploy directly
to Netlify or Vercel:

    const typewriterElement = document.getElementById('typewriter');

    function handleTypewriter() {
      const currentTitle = titles[titleIndex];
      if (isDeleting) {
        typewriterElement.textContent = currentTitle.substring(0, charIndex - 1);
        charIndex--;
        typingSpeed = 45;
      } else {
        typewriterElement.textContent = currentTitle.substring(0, charIndex + 1);
        charIndex++;
        typingSpeed = 95;
      }

      if (!isDeleting && charIndex === currentTitle.length) {
        typingSpeed = 2200; // Pause at full title
        isDeleting = true;
      } else if (isDeleting && charIndex === 0) {
        isDeleting = false;
        titleIndex = (titleIndex + 1) % titles.length;
        typingSpeed = 400; // Pause before typing next
      }

      setTimeout(handleTypewriter, typingSpeed);

    // --- 5. CANVAS PARTICLES STARFIELD (LIGHTWEIGHT) ---
    const canvas = document.getElementById('hero-particles');
    const ctx = canvas.getContext('2d');
    let particles = [];

    function resizeCanvas() {
      canvas.width = canvas.parentElement.offsetWidth;
      canvas.height = canvas.parentElement.offsetHeight;
    }
    resizeCanvas();
    window.addEventListener('resize', resizeCanvas);

    class Particle {
      constructor() {
        this.reset();
      }
      reset() {
        this.x = Math.random() * canvas.width;
        this.y = Math.random() * canvas.height;
        this.size = Math.random() * 1.6 + 0.3;
        this.speedX = (Math.random() - 0.5) * 0.4;
        this.speedY = (Math.random() - 0.5) * 0.4;
        this.alpha = Math.random() * 0.6 + 0.2;
      }
      update() {
        this.x += this.speedX;
        this.y += this.speedY;
        if (this.x < 0 || this.x > canvas.width || this.y < 0 || this.y > canvas.height) {
          this.reset();
        }
      }
      draw() {
        ctx.fillStyle = `rgba(214, 183, 122, ${this.alpha})`;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
      }
    }

    for (let i = 0; i < 48; i++) {
      particles.push(new Particle());
    }

    function animateParticles() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      particles.forEach(p => {
        p.update();
        p.draw();
      });
      requestAnimationFrame(animateParticles);
    }
    animateParticles();
